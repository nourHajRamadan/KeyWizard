String KWConfigInterface::getPasswordString(const char* key){
  begin(1);

  if(strcmp(key, "unlockpin") == 0){
    String unlockpin = passwordstrings.getString("unlockpin");
    if(unlockpin.isEmpty()){
      passwordstrings.putString("unlockpin", "000000");
    }
    return unlockpin;
  }
  if(strcmp(key, "ap-password") == 0){
    String ap_password = passwordstrings.getString("ap-password");
    if(ap_password.isEmpty()){
      passwordstrings.putString("ap-password", "00000000");
    }
    return ap_password;
  }
  size_t whatsLeft = passwordstrings.freeEntries();    
  Serial.printf("Entry slots left for passwords:  %lu \n", (unsigned long)whatsLeft);
  String value = passwordstrings.getString(key);
  return value.isEmpty()?"No value stored for that key":value;
}

bool KWConfigInterface::setPasswordString(const char* key, String value){
  begin(1);
  if(strcmp(key, "_KEYS_") == 0 || String(key).indexOf('|') != -1){
    Serial.println("Invalid Key!");
    return false;
  } 

  String keys = passwordstrings.getString("_KEYS_");
  if(!keys.length() || !keys.endsWith("|")) keys.concat('|');
  String token = "|" + String(key) + "|";
  if(keys.indexOf(token) != -1){
    Serial.println("Key already exists!");
    return false;
  }
  if(passwordstrings.putString(key,value)){
    keys.concat(String(key));
    keys.concat('|');
    passwordstrings.putString("_KEYS_", keys);
    return true;
  }
  return false;
}

bool KWConfigInterface::removePasswordString(const char* key){
  begin(1);
  if(passwordstrings.remove(key)){
    String keys = passwordstrings.getString("_KEYS_");
    if(!keys.length() || !keys.endsWith("|")) keys.concat('|');
    String token = "|" + String(key) + "|";
    int start = keys.indexOf(token)+1;
    if(start<1) return false;
    int end = start;
    while(end < keys.length()){
      if(keys[end]=='|')break;
      end++;
    }
    keys.remove(start,end-start+1);
    passwordstrings.putString("_KEYS_", keys);
    return true;
  }
  return false;
}

void KWCLI::processCommand(String command)
{
    command.trim();

    if(!unlocked){
        Serial.println("Enter unlock pin:");
        if(command == config->getPasswordString("unlockpin")){
            unlocked = true;
            Serial.println("KeyWizard unlocked");
            processCommand("help");
            return;
        }else{
            Serial.println("False pin. Try again..");
            return;            
        }
    }

    if(command.startsWith("help")){
        Serial.println("░█░█░█▀▀░█░█░█░█░▀█▀░▀▀█░█▀█░█▀▄░█▀▄");
        Serial.println("░█▀▄░█▀▀░░█░░█▄█░░█░░▄▀░░█▀█░█▀▄░█░█");
        Serial.println("░▀░▀░▀▀▀░░▀░░▀░▀░▀▀▀░▀▀▀░▀░▀░▀░▀░▀▀░");
        Serial.println("Commands:");
        Serial.println("help");
        Serial.println("get [OPTIONS] [KEY]");
        Serial.println("set [OPTIONS] [KEY] [VALUE]");
        Serial.println("remove all | [OPTIONS] [KEY]");
        Serial.println("OPTIONS: password | favourite | unlockpin | ap-password");
        Serial.println("exit");
        return;
    }

    if(command.startsWith("exit")){
        active = false;
        unlocked = false;
        Serial.println("Restarting KeyWizard...");
        ESP.restart();
        return;
    }

    if(command.startsWith("get ")){
        command = command.substring(4);

        if (command.startsWith("unlockpin")){
            Serial.println(config->getPasswordString("unlockpin"));
            return;
        }

        if (command.startsWith("ap-password")){
            Serial.println(config->getPasswordString("ap-password"));
            return;
        }

        if (command.startsWith("password ")){
            String key = command.substring(9);
            Serial.println(config->getPasswordString(key.c_str()));
            return;
        }

        if (command.startsWith("favourite ")){
            String key = command.substring(10);
            Serial.println(config->getFavouriteString(key.c_str()));
            return;
        }
    }
    
    if(command.startsWith("set ")){
        command = command.substring(4);

        if (command.startsWith("unlockpin ")){
            String value = command.substring(10);
            if(value.length()!=6){
                Serial.println("Unlockpin must be 6 characters long!");
                return;
            }

            bool ok = config->setPasswordString("unlockpin", value);
            if(ok){Serial.println("Setting unlockpin successful");}
            else{Serial.println("Setting unlockpin failed");}
            return;

        }

        if (command.startsWith("ap-password ")){
            String value = command.substring(12);

            bool ok = config->setPasswordString("ap-password", value);
            if(ok){Serial.println("Setting ap-password successful");}
            else{Serial.println("Setting ap-password failed");}
            return;

        }

        if (command.startsWith("password ")){
            String key = command.substring(9);
            int space = key.indexOf(' ');

            if(space==-1){ Serial.println("Invalid Command"); return;}

            String value = key.substring(space+1);
            key = key.substring(0,space);
            
            bool ok = config->setPasswordString(key.c_str(), value);
            if(ok){Serial.println("Setting password string successful");}
            else{Serial.println("Setting password string failed");}
            return;
        }

        if (command.startsWith("favourite ")){
            String key = command.substring(10);
            int space = key.indexOf(' ');

            if(space==-1){ Serial.println("Invalid Command"); return;}

            String value = key.substring(space+1);
            key = key.substring(0,space);
            
            bool ok = config->setFavouriteString(key.c_str(), value);
            if(ok){Serial.println("Setting favourite string successful");}
            else{Serial.println("Setting favourite string failed");}
            return;
        }
    }

    if(command.startsWith("remove ")){
        command = command.substring(7);

        if (command.startsWith("all")){
            bool ok = config->clearNVS();
            if(ok){Serial.println("Removing all configurations successful");}
            else{Serial.println("Removing all configurations failed");}
            return;
        }

        if (command.startsWith("unlockpin")){
            bool ok = config->removePasswordString("unlockpin");
            if(ok){Serial.println("Removing unlock successful");}
            else{Serial.println("Removing unlock failed");}
            return;
        }

        if (command.startsWith("ap-password")){
            bool ok = config->removePasswordString("ap-password");
            if(ok){Serial.println("Removing ap-password successful");}
            else{Serial.println("Removing ap-password failed");}
            return;
        }

        if (command.startsWith("password ")){
            String key = command.substring(9);
            bool ok = config->removePasswordString(key.c_str());
            if(ok){Serial.println("Removing password string successful");}
            else{Serial.println("Removing password string failed");}
            return;
        }

        if (command.startsWith("favourite ")){
            String key = command.substring(10);
            bool ok = config->removeFavouriteString(key.c_str());
            if(ok){Serial.println("Removing favourite string successful");}
            else{Serial.println("Removing favourite string failed");}
            return;
        }
    }

    Serial.println("Unknown command");
}
